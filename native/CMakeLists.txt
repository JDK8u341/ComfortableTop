cmake_minimum_required(VERSION 3.20)
project(ComfortableTopNative LANGUAGES CXX)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(ComfortableFrames 1 CONFIG REQUIRED)
pybind11_add_module(_ctui_top_native provider.cpp)
target_link_libraries(_ctui_top_native PRIVATE Comfortable::comfortable_cpu_frames)
target_compile_features(_ctui_top_native PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(_ctui_top_native PRIVATE Comfortable::comfortable_frames)
endif()
install(TARGETS _ctui_top_native LIBRARY DESTINATION native RUNTIME DESTINATION native)
if(MSVC)
    target_compile_options(_ctui_top_native PRIVATE /W4 /utf-8 /EHsc)
endif()
